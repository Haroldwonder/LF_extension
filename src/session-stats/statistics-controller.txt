import { 
  Controller, 
  Get, 
  Param, 
  Query, 
  UseGuards,

import { 
  ApiBearerAuth,

  ApiResponse,

} from '@nestjs/swagger';
import { JwtAuthGuard } from '../../auth/guards/jwt-auth.guard';

@ApiTags('statistics')
@Controller('statistics')
export class StatisticsController {
  private readonly logger = new Logger(StatisticsController.name);

  constructor(
    private readonly sessionAnalyticsService: SessionAnalyticsService,
    private readonly userStatisticsService: UserStatisticsService,
  ) {}

  @Get('sessions/:sessionId')
  @UseGuards(JwtAuthGuard)
  @ApiBearerAuth()
  @ApiOperation({ summary: 'Get complete session statistics' })
  @ApiParam({ name: 'sessionId', description: 'Session ID' })
  @ApiResponse({ status: 200, description: 'Session statistics retrieved successfully' })
  @ApiResponse({ status: 403, description: 'Forbidden - not owner of the session' })
  @ApiResponse({ status: 404, description: 'Session statistics not found' })
  async getSessionStatistics(@Param('sessionId') sessionId: string, @User() user) {
    try {
      const stats = await this.userStatisticsService.getSessionOwnership(sessionId);
      
      if (!stats) {
        throw new NotFoundException('Session statistics not found');
      }
      
      // Check if user is the owner of the session or has admin role
      if (stats.userId !== user.id && !user.roles?.includes('admin')) {
        throw new ForbiddenException('You do not have permission to access these statistics');
      }
      
      return stats;
    } catch (error) {
      this.logger.error(`Error fetching session statistics: ${error.message}`, error.stack);
      throw error;
    }
  }
